SET ProyectoDir TO $'''C:\\Users\\JUAN JOSE PEÑA\\Desktop\\reto tecnico'''
SET Reporte TO $'''%ProyectoDir%\\output\\reporte_solicitudes.csv'''
SET LogFlujo TO $'''%ProyectoDir%\\output\\log_flujo.txt'''
Scripting.RunDOSCommand.RunDOSCommand DOSCommandOrApplication: $'''\"%ProyectoDir%\\ejecutar.bat\"''' WorkingDirectory: ProyectoDir StandardOutput=> Salida StandardError=> Errores ExitCode=> CodigoSalida
File.IfFile.Exists File: Reporte
    IF CodigoSalida = 0 THEN
        File.WriteText File: LogFlujo TextToWrite: $'''%CurrentDateTime% - OK''' AppendNewLine: True IfFileExists: File.IfFileExists.Append Encoding: File.FileEncoding.UTF8
        Display.ShowMessageDialog.ShowMessage Title: $'''Automatizador RRHH''' Message: $'''Reporte generado correctamente''' Icon: Display.Icon.Information Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: False ButtonPressed=> Boton
    ELSE
        File.WriteText File: LogFlujo TextToWrite: $'''%CurrentDateTime% - FALLO codigo %CodigoSalida%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append Encoding: File.FileEncoding.UTF8
        Display.ShowMessageDialog.ShowMessage Title: $'''Automatizador RRHH''' Message: $'''La automatizacion fallo. Revisa output\\ultima_ejecucion.txt''' Icon: Display.Icon.Error Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: False ButtonPressed=> Boton
    END
ELSE
    File.WriteText File: LogFlujo TextToWrite: $'''%CurrentDateTime% - FALLO: no se genero el reporte''' AppendNewLine: True IfFileExists: File.IfFileExists.Append Encoding: File.FileEncoding.UTF8
    Display.ShowMessageDialog.ShowMessage Title: $'''Automatizador RRHH''' Message: $'''No se genero el reporte''' Icon: Display.Icon.Error Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: False ButtonPressed=> Boton
END
